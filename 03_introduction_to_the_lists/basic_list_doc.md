# Python Lists — базовая документация

## 1. Создание списка

my_list = [1, 2, 3] — создать список  
empty_list = [] — пустой список  

---

## 2. Обращение к элементам

my_list[0] — первый элемент  
my_list[1] — второй элемент  
my_list[-1] — последний элемент  
my_list[-2] — предпоследний элемент  

---

## 3. Добавление элементов

my_list.append(x) — добавить элемент в конец списка  

my_list.insert(index, x) — вставить элемент по индексу  

---

## 4. Удаление элементов

del my_list[index] — удалить элемент по индексу  

my_list.pop() — удалить последний элемент и вернуть его  

my_list.pop(index) — удалить элемент по индексу и вернуть его  

my_list.remove(x) — удалить первый найденный элемент со значением x  

---

## 5. Сортировка

sorted(my_list) — вернуть новый отсортированный список (оригинал не меняется)  

sorted(my_list, reverse=True) — вернуть новый список в обратном порядке  

my_list.sort() — отсортировать список на месте (изменяет оригинал)  

my_list.sort(reverse=True) — сортировка на месте в обратном порядке  

---

## 6. Переворот списка

my_list.reverse() — перевернуть список на месте (изменяет оригинал)  

reversed(my_list) — вернуть итератор (нужно обернуть в list())

list(reversed(my_list)) — получить перевёрнутый список  

---

## 7. Длина списка

len(my_list) — количество элементов в списке  

---

## 8. Копирование списка

new_list = my_list[:] — копия списка  

new_list = list(my_list) — ещё один способ копирования  

---

## 9. Перебор списка (базово)

for item in my_list: — пройтись по каждому элементу  

---

## 10. Важные различия

sorted(my_list) — новый список  
my_list.sort() — меняет оригинал  

reversed(my_list) — итератор  
my_list.reverse() — меняет оригинал  

append() — добавляет в конец  
insert() — вставляет по индексу  

pop() — удаляет и возвращает  
remove() — удаляет по значению  

---

## 11. Частые ошибки

print(my_list.sort()) — выведет None  

print(my_list.reverse()) — выведет None  

reversed(my_list) — не список, а итератор  

my_list[100] — ошибка, если индекс вне диапазона  

---

## 12. Мини-шпаргалка

создать список → []  
взять элемент → list[index]  
добавить → append()  
вставить → insert()  
удалить по индексу → del / pop()  
удалить по значению → remove()  
отсортировать → sort() / sorted()  
перевернуть → reverse() / reversed()  
длина → len()
